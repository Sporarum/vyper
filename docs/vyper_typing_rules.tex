\begin{abstract}
This document presents the formal typing rules for the Vyper smart contract programming language,
extracted from the compiler implementation. Vyper is a statically typed, Pythonic language
targeting the Ethereum Virtual Machine (EVM). These rules follow standard notation from
Pierce's \emph{Types and Programming Languages}~\cite{pierce2002types} and effect system
literature~\cite{nielson1999type}.
\end{abstract}

\section{Type Syntax}
%==============================================================================

\subsection{Base Types}

\begin{align*}
\tau ::= \quad & \kw{bool} && \text{Boolean} \\
\mid \quad & \kw{int}N && \text{Signed integer } (N \in \{8,16,\ldots,256\}) \\
\mid \quad & \kw{uint}N && \text{Unsigned integer } (N \in \{8,16,\ldots,256\}) \\
\mid \quad & \kw{decimal} && \text{Fixed-point decimal (168 bits, 10 decimals)} \\
\mid \quad & \kw{address} && \text{Ethereum address (160 bits)} \\
\mid \quad & \kw{bytes}M && \text{Fixed-size byte array } (M \in \{1,\ldots,32\}) \\
\mid \quad & \kw{Bytes}[n] && \text{Dynamic byte array (max length } n \text{)} \\
\mid \quad & \kw{String}[n] && \text{Dynamic string (max length } n \text{)}
\end{align*}

\subsection{Composite Types}

\begin{align*}
\tau ::= \quad & \tau[n] && \text{Static array of length } n \\
\mid \quad & \kw{DynArray}[\tau, n] && \text{Dynamic array (max length } n \text{)} \\
\mid \quad & \kw{HashMap}[\kappa, \tau] && \text{Mapping from } \kappa \text{ to } \tau \\
\mid \quad & (\tau_1, \ldots, \tau_n) && \text{Tuple} \\
\mid \quad & \kw{struct}\ S && \text{User-defined struct} \\
\mid \quad & \kw{flag}\ F && \text{User-defined flag (enum)} \\
\mid \quad & \kw{event}\ E && \text{User-defined event} \\
\mid \quad & \kw{interface}\ I && \text{Interface type}
\end{align*}

%==============================================================================
\section{Typing Contexts and Judgments}
%==============================================================================

\subsection{Type Environment}

\[
\Gamma ::= \emptyset \mid \Gamma, x : \tau
\]

\subsection{Data Locations}

\[
\ell ::= \kw{storage} \mid \kw{transient} \mid \kw{memory} \mid \kw{calldata} \mid \kw{code}
\]

\subsection{Modifiability (for variables)}

\[
\mu ::= \kw{constant} < \kw{runtime\_constant} < \kw{modifiable}
\]

\subsection{State Mutability (for functions)}

\[
m ::= \kw{pure} < \kw{view} < \kw{nonpayable} < \kw{payable}
\]

\subsection{Judgment Forms}

We use the following judgment forms, following standard conventions~\cite{pierce2002types}:

\begin{center}
\begin{tabular}{ll}
$\Gamma; m \vdash e : \tau$ & Expression $e$ has type $\tau$ in context $\Gamma$ under mutability $m$ \\
$\Gamma; m \vdash s \Rightarrow \Gamma'$ & Statement $s$ transforms $\Gamma$ to $\Gamma'$ under mutability $m$ \\
$\subtype{\tau_1}{\tau_2}$ & Type $\tau_1$ is a subtype of $\tau_2$ \\
\end{tabular}
\end{center}

The notation $\Gamma; m \vdash \cdot$ threads the function mutability $m$ through all judgments,
enabling uniform enforcement of state access constraints.

%==============================================================================
\section{Subtyping Rules}
%==============================================================================

\fbox{$\subtype{\tau_1}{\tau_2}$} \quad ($\tau_1$ is a subtype of $\tau_2$)

\subsection{Structural Rules}

\begin{mathpar}
\inferrule*[right=S-Refl]
  { }
  {\subtype{\tau}{\tau}}

\inferrule*[right=S-Trans]
  {\subtype{\tau_1}{\tau_2} \\ \subtype{\tau_2}{\tau_3}}
  {\subtype{\tau_1}{\tau_3}}
\end{mathpar}

\subsection{Numeric Types (No Implicit Coercion)}

Vyper requires exact type matching for numeric types---there are no implicit widening conversions.

\begin{mathpar}
\inferrule*[right=S-Int]
  { }
  {\subtype{\kw{int}N}{\kw{int}N}}

\inferrule*[right=S-Uint]
  { }
  {\subtype{\kw{uint}N}{\kw{uint}N}}

\inferrule*[right=S-Decimal]
  { }
  {\subtype{\kw{decimal}}{\kw{decimal}}}
\end{mathpar}

\subsection{Array Subtyping}

Static arrays are \emph{invariant} in both element type and length:

\begin{mathpar}
\inferrule*[right=S-SArray]
  {\tau_1 = \tau_2 \\ n_1 = n_2}
  {\subtype{\tau_1[n_1]}{\tau_2[n_2]}}
\end{mathpar}

Dynamic arrays are \emph{covariant} in element type and \emph{contravariant} in max length
(a smaller-capacity array can be used where a larger one is expected):

\begin{mathpar}
\inferrule*[right=S-DynArray]
  {\subtype{\tau_1}{\tau_2} \\ n_1 \leq n_2}
  {\subtype{\kw{DynArray}[\tau_1, n_1]}{\kw{DynArray}[\tau_2, n_2]}}
\end{mathpar}

\subsection{Bytestring Subtyping}

Bytestrings with smaller max length are subtypes of those with larger max length:

\begin{mathpar}
\inferrule*[right=S-Bytes]
  {n_1 \leq n_2}
  {\subtype{\kw{Bytes}[n_1]}{\kw{Bytes}[n_2]}}

\inferrule*[right=S-String]
  {n_1 \leq n_2}
  {\subtype{\kw{String}[n_1]}{\kw{String}[n_2]}}
\end{mathpar}

\subsection{Special Types}

User-defined types use \emph{nominal} equality (identity comparison):

\begin{mathpar}
\inferrule*[right=S-Self]
  { }
  {\subtype{\kw{self}}{\kw{address}}}

\inferrule*[right=S-Struct]
  {S_1 \equiv S_2}
  {\subtype{\kw{struct}\ S_1}{\kw{struct}\ S_2}}

\inferrule*[right=S-Flag]
  {F_1 \equiv F_2}
  {\subtype{\kw{flag}\ F_1}{\kw{flag}\ F_2}}
\end{mathpar}

%==============================================================================
\section{Type Predicates}
%==============================================================================

The following predicates define validity conditions for type constructors.

\subsection{Valid Static Array Element Types}

The predicate $\validelem{\tau}$ determines if $\tau$ can be an element of a static array:

\begin{mathpar}
\inferrule*[right=VE-Prim]
  {\tau \in \{\kw{bool}, \kw{int}N, \kw{uint}N, \kw{decimal}, \kw{address}, \kw{bytes}M\}}
  {\validelem{\tau}}

\inferrule*[right=VE-Struct]
  { }
  {\validelem{\kw{struct}\ S}}

\inferrule*[right=VE-DArray]
  { }
  {\validelem{\kw{DynArray}[\tau, n]}}

\inferrule*[right=VE-SArray]
  {\validelem{\tau}}
  {\validelem{\tau[n]}}

\inferrule*[right=VE-Interface]
  { }
  {\validelem{\kw{interface}\ I}}
\end{mathpar}

Note: $\kw{Bytes}[n]$, $\kw{String}[n]$, $\kw{flag}$, $\kw{HashMap}$, and $\kw{event}$ are \emph{not} valid static array elements.

\subsection{Valid Dynamic Array Element Types}

The predicate $\validdarray{\tau}$ determines if $\tau$ can be an element of a dynamic array:

\begin{mathpar}
\inferrule*[right=VD-Elem]
  {\validelem{\tau}}
  {\validdarray{\tau}}

\inferrule*[right=VD-Bytes]
  { }
  {\validdarray{\kw{Bytes}[n]}}

\inferrule*[right=VD-String]
  { }
  {\validdarray{\kw{String}[n]}}

\inferrule*[right=VD-Flag]
  { }
  {\validdarray{\kw{flag}\ F}}
\end{mathpar}

\subsection{Valid HashMap Key Types}

The predicate $\validkey{\kappa}$ determines if $\kappa$ can be a HashMap key:

\begin{mathpar}
\inferrule*[right=VK-Prim]
  {\kappa \in \{\kw{bool}, \kw{int}N, \kw{uint}N, \kw{address}, \kw{bytes}M\}}
  {\validkey{\kappa}}

\inferrule*[right=VK-Bytes]
  { }
  {\validkey{\kw{Bytes}[n]}}

\inferrule*[right=VK-String]
  { }
  {\validkey{\kw{String}[n]}}

\inferrule*[right=VK-Flag]
  { }
  {\validkey{\kw{flag}\ F}}
\end{mathpar}

%==============================================================================
\section{Expression Typing Rules}
%==============================================================================

\fbox{$\Gamma; m \vdash e : \tau$} \quad (expression $e$ has type $\tau$ in context $\Gamma$ under mutability $m$)

\subsection{Literals}

\begin{mathpar}
\inferrule*[right=T-True]
  { }
  {\Gamma; m \vdash \kw{True} : \kw{bool}}

\inferrule*[right=T-False]
  { }
  {\Gamma; m \vdash \kw{False} : \kw{bool}}

\inferrule*[right=T-IntLit]
  {-2^{N-1} \leq n \leq 2^{N-1} - 1}
  {\Gamma; m \vdash n : \kw{int}N}

\inferrule*[right=T-UintLit]
  {0 \leq n \leq 2^N - 1}
  {\Gamma; m \vdash n : \kw{uint}N}

\inferrule*[right=T-DecLit]
  {-2^{167}/10^{10} \leq d \leq (2^{167}-1)/10^{10}}
  {\Gamma; m \vdash d : \kw{decimal}}

\inferrule*[right=T-Addr]
  {\text{checksummed}(a)}
  {\Gamma; m \vdash \kw{0x}a : \kw{address}}

\inferrule*[right=T-BytesM]
  {|\mathit{hex}| = M \\ 1 \leq M \leq 32}
  {\Gamma; m \vdash \kw{0x}\mathit{hex} : \kw{bytes}M}

\inferrule*[right=T-BytesLit]
  {|s| \leq n}
  {\Gamma; m \vdash \kw{b"}s\kw{"} : \kw{Bytes}[n]}

\inferrule*[right=T-StrLit]
  {|s| \leq n}
  {\Gamma; m \vdash \kw{"}s\kw{"} : \kw{String}[n]}
\end{mathpar}

\subsection{Variables}

\begin{mathpar}
\inferrule*[right=T-Var]
  {(x : \tau) \in \Gamma}
  {\Gamma; m \vdash x : \tau}
\end{mathpar}

\subsection{State Variables with Mutability Constraints}

\begin{mathpar}
\inferrule*[right=T-StateRead]
  {(x : \tau) \in \Gamma \\ \ell(x) \in \{\kw{storage}, \kw{transient}\} \\ m \neq \kw{pure}}
  {\Gamma; m \vdash x : \tau}

\inferrule*[right=T-EnvRead]
  {v \in \{\kw{block.number}, \kw{msg.sender}, \ldots\} \\ m \neq \kw{pure}}
  {\Gamma; m \vdash v : \tau_v}

\inferrule*[right=T-MsgValue]
  {m = \kw{payable}}
  {\Gamma; m \vdash \kw{msg.value} : \kw{uint256}}
\end{mathpar}

\subsection{Boolean Operations}

\begin{mathpar}
\inferrule*[right=T-And]
  {\Gamma; m \vdash e_1 : \kw{bool} \\ \Gamma; m \vdash e_2 : \kw{bool}}
  {\Gamma; m \vdash e_1\ \kw{and}\ e_2 : \kw{bool}}

\inferrule*[right=T-Or]
  {\Gamma; m \vdash e_1 : \kw{bool} \\ \Gamma; m \vdash e_2 : \kw{bool}}
  {\Gamma; m \vdash e_1\ \kw{or}\ e_2 : \kw{bool}}

\inferrule*[right=T-Not]
  {\Gamma; m \vdash e : \kw{bool}}
  {\Gamma; m \vdash \kw{not}\ e : \kw{bool}}
\end{mathpar}

\subsection{Arithmetic Operations}

Let $\mathit{NumericT} = \{\kw{int}N, \kw{uint}N, \kw{decimal}\}$ and
$\mathit{IntegerT} = \{\kw{int}N, \kw{uint}N\}$.

\begin{mathpar}
\inferrule*[right=T-BinArith]
  {\Gamma; m \vdash e_1 : \tau \\ \Gamma; m \vdash e_2 : \tau \\ \tau \in \mathit{NumericT} \\
   \oplus \in \{+, -, *, /\!/, \%\}}
  {\Gamma; m \vdash e_1 \oplus e_2 : \tau}

\inferrule*[right=T-DecDiv]
  {\Gamma; m \vdash e_1 : \kw{decimal} \\ \Gamma; m \vdash e_2 : \kw{decimal}}
  {\Gamma; m \vdash e_1\ /\ e_2 : \kw{decimal}}

\inferrule*[right=T-Neg]
  {\Gamma; m \vdash e : \tau \\ \tau \in \{\kw{int}N, \kw{decimal}\}}
  {\Gamma; m \vdash -e : \tau}
\end{mathpar}

\subsection{Exponentiation}

Exponentiation requires at least one operand to be a compile-time constant for bounds checking:

\begin{mathpar}
\inferrule*[right=T-PowLit]
  {\Gamma; m \vdash e_1 : \tau \\ \Gamma; m \vdash e_2 : \tau \\
   \tau \in \mathit{IntegerT} \\ \text{isLiteral}(e_1) \lor \text{isLiteral}(e_2)}
  {\Gamma; m \vdash e_1\ \kw{**}\ e_2 : \tau}
\end{mathpar}

\subsection{Bitwise Operations}

\begin{mathpar}
\inferrule*[right=T-BitBin]
  {\Gamma; m \vdash e_1 : \tau \\ \Gamma; m \vdash e_2 : \tau \\
   \tau \in \{\kw{uint}N, \kw{bytes}M, \kw{flag}\ F\} \\ \odot \in \{\&, |, \wedge\}}
  {\Gamma; m \vdash e_1\ \odot\ e_2 : \tau}

\inferrule*[right=T-BitNot]
  {\Gamma; m \vdash e : \tau \\ \tau \in \{\kw{uint256}, \kw{bytes32}, \kw{flag}\ F\}}
  {\Gamma; m \vdash \mathord{\sim} e : \tau}
\end{mathpar}

\subsection{Shift Operations}

Shifts are only allowed on 256-bit types:

\begin{mathpar}
\inferrule*[right=T-Shift]
  {\Gamma; m \vdash e_1 : \tau \\ \Gamma; m \vdash e_2 : \kw{uint}N \\
   \tau \in \{\kw{uint256}, \kw{int256}, \kw{bytes32}\} \\ \diamond \in \{\ll, \gg\}}
  {\Gamma; m \vdash e_1\ \diamond\ e_2 : \tau}
\end{mathpar}

\subsection{Comparisons}

\begin{mathpar}
\inferrule*[right=T-NumCmp]
  {\Gamma; m \vdash e_1 : \tau \\ \Gamma; m \vdash e_2 : \tau \\
   \tau \in \mathit{NumericT} \\ \bowtie\ \in \{<, \leq, >, \geq\}}
  {\Gamma; m \vdash e_1\ \bowtie\ e_2 : \kw{bool}}

\inferrule*[right=T-Eq]
  {\Gamma; m \vdash e_1 : \tau \\ \Gamma; m \vdash e_2 : \tau}
  {\Gamma; m \vdash e_1 == e_2 : \kw{bool}}

\inferrule*[right=T-NEq]
  {\Gamma; m \vdash e_1 : \tau \\ \Gamma; m \vdash e_2 : \tau}
  {\Gamma; m \vdash e_1\ !=\ e_2 : \kw{bool}}
\end{mathpar}

\subsection{Membership Tests}

\begin{mathpar}
\inferrule*[right=T-InArray]
  {\Gamma; m \vdash e : \tau \\ \Gamma; m \vdash a : \tau[n] \lor \Gamma; m \vdash a : \kw{DynArray}[\tau, n]}
  {\Gamma; m \vdash e\ \kw{in}\ a : \kw{bool}}

\inferrule*[right=T-InFlag]
  {\Gamma; m \vdash e : \kw{flag}\ F \\ \Gamma; m \vdash f : \kw{flag}\ F}
  {\Gamma; m \vdash e\ \kw{in}\ f : \kw{bool}}
\end{mathpar}

\subsection{Subscript Access}

\begin{mathpar}
\inferrule*[right=T-SArrayIdx]
  {\Gamma; m \vdash e : \tau[n] \\ \Gamma; m \vdash i : \tau_i \\ \tau_i \in \mathit{IntegerT} \\ 0 \leq i < n}
  {\Gamma; m \vdash e[i] : \tau}

\inferrule*[right=T-DArrayIdx]
  {\Gamma; m \vdash e : \kw{DynArray}[\tau, n] \\ \Gamma; m \vdash i : \tau_i \\ \tau_i \in \mathit{IntegerT}}
  {\Gamma; m \vdash e[i] : \tau}

\inferrule*[right=T-MapIdx]
  {\Gamma; m \vdash h : \kw{HashMap}[\kappa, \tau] \\ \Gamma; m \vdash k : \kappa}
  {\Gamma; m \vdash h[k] : \tau}

\inferrule*[right=T-TupleIdx]
  {\Gamma; m \vdash t : (\tau_0, \ldots, \tau_{n-1}) \\ 0 \leq i < n \\ \text{isLiteral}(i)}
  {\Gamma; m \vdash t[i] : \tau_i}
\end{mathpar}

\subsection{Attribute Access}

\begin{mathpar}
\inferrule*[right=T-StructAttr]
  {\Gamma; m \vdash e : \kw{struct}\ S \\ (f : \tau) \in S}
  {\Gamma; m \vdash e.f : \tau}

\inferrule*[right=T-AddrAttr]
  {\Gamma; m \vdash e : \kw{address} \\ m \neq \kw{pure} \\
   a \in \{\kw{balance} : \kw{uint256}, \kw{codesize} : \kw{uint256},
           \kw{codehash} : \kw{bytes32}, \kw{is\_contract} : \kw{bool}\}}
  {\Gamma; m \vdash e.a : \tau_a}
\end{mathpar}

\subsection{Composite Literals}

\begin{mathpar}
\inferrule*[right=T-ArrayLit]
  {\Gamma; m \vdash e_i : \tau \text{ for all } i \in 1..n}
  {\Gamma; m \vdash [e_1, \ldots, e_n] : \tau[n]}

\inferrule*[right=T-Tuple]
  {\Gamma; m \vdash e_i : \tau_i \text{ for all } i \in 1..n}
  {\Gamma; m \vdash (e_1, \ldots, e_n) : (\tau_1, \ldots, \tau_n)}
\end{mathpar}

\subsection{Struct Construction}

\begin{mathpar}
\inferrule*[right=T-StructCtor]
  {\kw{struct}\ S = \{f_1 : \tau_1, \ldots, f_n : \tau_n\} \\
   \Gamma; m \vdash e_i : \tau_i' \\ \subtype{\tau_i'}{\tau_i} \text{ for all } i}
  {\Gamma; m \vdash S(f_1 = e_1, \ldots, f_n = e_n) : \kw{struct}\ S}
\end{mathpar}

\subsection{Conditional Expression}

\begin{mathpar}
\inferrule*[right=T-IfExp]
  {\Gamma; m \vdash e_c : \kw{bool} \\ \Gamma; m \vdash e_1 : \tau \\ \Gamma; m \vdash e_2 : \tau}
  {\Gamma; m \vdash e_1\ \kw{if}\ e_c\ \kw{else}\ e_2 : \tau}
\end{mathpar}

\subsection{Type Conversion}

\begin{mathpar}
\inferrule*[right=T-Convert]
  {\Gamma; m \vdash e : \tau_1 \\ \convertible{\tau_1}{\tau_2}}
  {\Gamma; m \vdash \kw{convert}(e, \tau_2) : \tau_2}
\end{mathpar}

\subsection{Dynamic Array Methods}

\begin{mathpar}
\inferrule*[right=T-Append]
  {\Gamma; m \vdash a : \kw{DynArray}[\tau, n] \\ \Gamma; m \vdash e : \tau' \\ \subtype{\tau'}{\tau} \\
   \mu(a) = \kw{modifiable}}
  {\Gamma; m \vdash a.\kw{append}(e) : ()}

\inferrule*[right=T-Pop]
  {\Gamma; m \vdash a : \kw{DynArray}[\tau, n] \\ \mu(a) = \kw{modifiable}}
  {\Gamma; m \vdash a.\kw{pop}() : \tau}
\end{mathpar}

%==============================================================================
\section{Function Typing Rules}
%==============================================================================

\subsection{Function Types}

Function types include visibility and mutability:
\[
\sigma ::= (\tau_1, \ldots, \tau_n) \xrightarrow{v, m} \tau_r
\]
where $v \in \{\kw{internal}, \kw{external}, \kw{deploy}\}$ and $m \in \{\kw{pure}, \kw{view}, \kw{nonpayable}, \kw{payable}\}$.

\subsection{Function Definition}

\begin{mathpar}
\inferrule*[right=T-FunDef]
  {\Gamma, x_1 : \tau_1, \ldots, x_n : \tau_n; m_f \vdash \mathit{body} \Rightarrow \Gamma' \\
   \text{returns}(\mathit{body}, \tau_r)}
  {\Gamma \vdash \kw{@}v\ \kw{@}m_f\ \kw{def}\ f(\overline{x : \tau}) \to \tau_r :
   (\overline{\tau}) \xrightarrow{v, m_f} \tau_r}
\end{mathpar}

\subsection{Internal Function Call}

\begin{mathpar}
\inferrule*[right=T-InternalCall]
  {(f : (\overline{\tau}) \xrightarrow{\kw{internal}, m_f} \tau_r) \in \Gamma \\
   \Gamma; m \vdash e_i : \tau_i' \\ \subtype{\tau_i'}{\tau_i} \text{ for all } i \\
   m_f \leq m \lor m \geq \kw{nonpayable}}
  {\Gamma; m \vdash f(\overline{e}) : \tau_r}
\end{mathpar}

\subsection{External Function Calls}

External calls require explicit \kw{extcall} or \kw{staticcall} keywords:

\begin{mathpar}
\inferrule*[right=T-ExtCall]
  {\Gamma; m \vdash e : \kw{interface}\ I \\
   (f : (\overline{\tau}) \xrightarrow{\kw{external}, m_f} \tau_r) \in I \\
   m_f \geq \kw{nonpayable} \\
   \Gamma; m \vdash e_i : \tau_i' \\ \subtype{\tau_i'}{\tau_i} \text{ for all } i \\
   m \geq \kw{nonpayable}}
  {\Gamma; m \vdash \kw{extcall}\ e.f(\overline{e}) : \tau_r}

\inferrule*[right=T-StaticCall]
  {\Gamma; m \vdash e : \kw{interface}\ I \\
   (f : (\overline{\tau}) \xrightarrow{\kw{external}, m_f} \tau_r) \in I \\
   m_f \in \{\kw{view}, \kw{pure}\} \\
   \Gamma; m \vdash e_i : \tau_i' \\ \subtype{\tau_i'}{\tau_i} \text{ for all } i}
  {\Gamma; m \vdash \kw{staticcall}\ e.f(\overline{e}) : \tau_r}
\end{mathpar}

%==============================================================================
\section{Statement Typing Rules}
%==============================================================================

\fbox{$\Gamma; m \vdash s \Rightarrow \Gamma'$} \quad (statement $s$ transforms $\Gamma$ to $\Gamma'$ under mutability $m$)

The arrow notation $\Rightarrow$ follows the convention for small-step semantics and effect
systems~\cite{wright1994syntactic}, indicating that execution of $s$ may extend the typing context.

\subsection{Variable Declaration}

\begin{mathpar}
\inferrule*[right=T-VarDecl]
  {\Gamma; m \vdash e : \tau' \\ \subtype{\tau'}{\tau}}
  {\Gamma; m \vdash (x : \tau = e) \Rightarrow \Gamma, x : \tau}
\end{mathpar}

\subsection{Assignment}

\begin{mathpar}
\inferrule*[right=T-Assign]
  {(x : \tau) \in \Gamma \\ \Gamma; m \vdash e : \tau' \\
   \subtype{\tau'}{\tau} \\ \mu(x) = \kw{modifiable} \\
   \ell(x) \in \{\kw{storage}, \kw{transient}\} \Rightarrow m \geq \kw{nonpayable}}
  {\Gamma; m \vdash (x = e) \Rightarrow \Gamma}

\inferrule*[right=T-AugAssign]
  {(x : \tau) \in \Gamma \\ \Gamma; m \vdash e : \tau \\ \tau \in \mathit{NumericT} \\
   \mu(x) = \kw{modifiable} \\
   \ell(x) \in \{\kw{storage}, \kw{transient}\} \Rightarrow m \geq \kw{nonpayable}}
  {\Gamma; m \vdash (x\ \oplus\!=\ e) \Rightarrow \Gamma}
\end{mathpar}

\subsection{Control Flow}

\begin{mathpar}
\inferrule*[right=T-If]
  {\Gamma; m \vdash e : \kw{bool} \\
   \Gamma; m \vdash s_1 \Rightarrow \Gamma_1 \\
   \Gamma; m \vdash s_2 \Rightarrow \Gamma_2}
  {\Gamma; m \vdash (\kw{if}\ e\kw{:}\ s_1\ \kw{else:}\ s_2) \Rightarrow \Gamma}

\inferrule*[right=T-ForRange]
  {\Gamma; m \vdash e_1 : \tau \\ \Gamma; m \vdash e_2 : \tau \\ \tau \in \mathit{IntegerT} \\
   \Gamma, i : \tau; m \vdash s \Rightarrow \Gamma'}
  {\Gamma; m \vdash (\kw{for}\ i : \tau\ \kw{in range}(e_1, e_2)\kw{:}\ s) \Rightarrow \Gamma}

\inferrule*[right=T-ForArray]
  {\Gamma; m \vdash a : \tau[n] \lor \Gamma; m \vdash a : \kw{DynArray}[\tau, n] \\
   \Gamma, x : \tau; m \vdash s \Rightarrow \Gamma'}
  {\Gamma; m \vdash (\kw{for}\ x : \tau\ \kw{in}\ a\kw{:}\ s) \Rightarrow \Gamma}
\end{mathpar}

\subsection{Return}

\begin{mathpar}
\inferrule*[right=T-Return]
  {\Gamma; m \vdash e : \tau' \\ \subtype{\tau'}{\tau_r}}
  {\Gamma; m \vdash \kw{return}\ e \Rightarrow \bot}

\inferrule*[right=T-ReturnVoid]
  {\tau_r = ()}
  {\Gamma; m \vdash \kw{return} \Rightarrow \bot}
\end{mathpar}

\subsection{Assert}

Assert can optionally take a revert message:

\begin{mathpar}
\inferrule*[right=T-Assert]
  {\Gamma; m \vdash e : \kw{bool}}
  {\Gamma; m \vdash \kw{assert}\ e \Rightarrow \Gamma}

\inferrule*[right=T-AssertMsg]
  {\Gamma; m \vdash e : \kw{bool} \\ \Gamma; m \vdash \mathit{msg} : \kw{String}[n] \\ n \leq 1024}
  {\Gamma; m \vdash \kw{assert}\ e,\ \mathit{msg} \Rightarrow \Gamma}

\inferrule*[right=T-AssertUnreach]
  {\Gamma; m \vdash e : \kw{bool}}
  {\Gamma; m \vdash \kw{assert}\ e,\ \kw{UNREACHABLE} \Rightarrow \Gamma}
\end{mathpar}

\subsection{Raise}

\begin{mathpar}
\inferrule*[right=T-Raise]
  {\Gamma; m \vdash e : \kw{String}[n] \\ n \leq 1024}
  {\Gamma; m \vdash \kw{raise}\ e \Rightarrow \bot}

\inferrule*[right=T-RaiseUnreach]
  { }
  {\Gamma; m \vdash \kw{raise}\ \kw{UNREACHABLE} \Rightarrow \bot}
\end{mathpar}

\subsection{Event Logging}

\begin{mathpar}
\inferrule*[right=T-Log]
  {\kw{event}\ E = \{a_1 : \tau_1, \ldots, a_n : \tau_n\} \\
   \Gamma; m \vdash e_i : \tau_i' \\ \subtype{\tau_i'}{\tau_i} \text{ for all } i \\
   m \geq \kw{nonpayable}}
  {\Gamma; m \vdash \kw{log}\ E(a_1 = e_1, \ldots, a_n = e_n) \Rightarrow \Gamma}
\end{mathpar}

\section{Type Conversion Rules}
%==============================================================================

The predicate $\convertible{\tau_1}{\tau_2}$ defines valid explicit conversions:

\subsection{Integer Conversions}

\begin{mathpar}
\inferrule*[right=Conv-IntInt]
  {\tau_1 \in \mathit{IntegerT} \\ \tau_2 \in \mathit{IntegerT}}
  {\convertible{\tau_1}{\tau_2}}

\inferrule*[right=Conv-IntDec]
  {\tau \in \mathit{IntegerT}}
  {\convertible{\tau}{\kw{decimal}}}

\inferrule*[right=Conv-DecInt]
  {\tau \in \mathit{IntegerT}}
  {\convertible{\kw{decimal}}{\tau}}
\end{mathpar}

\subsection{Byte Conversions}

\begin{mathpar}
\inferrule*[right=Conv-BytesInt]
  { }
  {\convertible{\kw{bytes}M}{\kw{int}N} \land \convertible{\kw{bytes}M}{\kw{uint}N}}

\inferrule*[right=Conv-IntBytes]
  { }
  {\convertible{\kw{int}N}{\kw{bytes}M} \land \convertible{\kw{uint}N}{\kw{bytes}M}}
\end{mathpar}

\subsection{Address Conversions}

\begin{mathpar}
\inferrule*[right=Conv-AddrInt]
  { }
  {\convertible{\kw{address}}{\kw{uint160}} \land \convertible{\kw{uint160}}{\kw{address}}}

\inferrule*[right=Conv-AddrBytes]
  { }
  {\convertible{\kw{address}}{\kw{bytes20}} \land \convertible{\kw{bytes20}}{\kw{address}}}
\end{mathpar}

\subsection{Boolean Conversions}

\begin{mathpar}
\inferrule*[right=Conv-ToBool]
  {\tau \in \mathit{IntegerT} \cup \{\kw{decimal}, \kw{bytes}M, \kw{address}\}}
  {\convertible{\tau}{\kw{bool}}}
\end{mathpar}

\subsection{Flag Conversions}

\begin{mathpar}
\inferrule*[right=Conv-Flag]
  { }
  {\convertible{\kw{flag}\ F}{\kw{uint256}} \land \convertible{\kw{uint256}}{\kw{flag}\ F}}
\end{mathpar}

\subsection{Type-Location Restrictions}

\begin{mathpar}
\inferrule*[right=L-HashMap]

  {\kw{HashMap}[\kappa, \tau]\ \text{valid at}\ \ell}

  {\text{params}(f)\ \text{have location}\ \kw{calldata}}

  {\text{params}(f)\ \text{have location}\ \kw{memory}}

%==============================================================================
\section{Summary}
%==============================================================================

The Vyper type system is characterized by:

\begin{enumerate}
  \item \textbf{No implicit coercions}: Types must match exactly or through explicit subtyping
  \item \textbf{Nominal typing}: User-defined types use identity comparison
  \item \textbf{Structural subtyping}: Bytestrings and dynamic arrays by max length
  \item \textbf{Effect tracking}: Function mutability ($m$) threads through all judgments
  \item \textbf{Location tracking}: Variables have associated storage locations
  \item \textbf{Explicit conversions}: Via \kw{convert()} with bounds checking
  \item \textbf{Static bounds}: Integer overflow checked at compile-time where possible
\end{enumerate}
